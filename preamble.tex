  % #1 -- отступ слева, #2 -- запись (\numberline{..}Заголовок), #3 -- страница
  \vskip \z@ \@plus.2\p@
  {\leftskip #1\relax \rightskip \@tocrmarg \parfillskip -\rightskip

   \leavevmode \normalsize

   \nobreak \hb@xt@\@pnumwidth{\hfil\normalcolor #3}\par}}
